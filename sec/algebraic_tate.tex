\section{The algebraic Tate diagonal}

\begin{definition} We denote by $T_p$ the functor
\[
T_p\coloneqq (-\otimes \overset{p}{\cdots}\otimes -)^{t\cyc_p} \colon \mathcal D(\Fp)\to \mathcal D(\Fp).
\]
Explicitly, given $V$ in $\mathcal D(\Fp)$, $T_pV$ is the $\cyc_p$-Tate construction of the chain complex of $\Fp[\cyc_p]$-modules $V^{\otimes p}$, whose $\cyc_p$-action is given by permuting the factors.
\end{definition}

\begin{example}
Taking $V=\Fp$, we have $T_p(\Fp)=(\Fp^{\otimes p})^{t\cyc_p}\cong \Fp^{t\cyc_p}$. Its homology is precisely the Tate cohomology of the cyclic group with trivial coefficients $\hat H^*(\cyc_p;\Fp)$, and is given by
\[
\hat H^i(\cyc_p;\Fp)\cong H_i(\widehat W_p\otimes_{\cyc_p}\Fp)=H_i((\widehat W_p)_{\cyc_p})\cong \Fp\{\hat e_i\}.
\]
Therefore, $H_*(T_p\Fp)$ is a free one-dimensional $\Fp[u,u^{-1}]$-module with action
\[
u\cdot \hat e_i\coloneqq \hat e_{i+1}.
\]
\end{example}
\begin{lemma} $T_p$ is an exact functor.
\end{lemma}%immediate from Nikolaus-Scholze?

\begin{definition} The \defn{algebraic Tate diagonal} is the natural transformation 
\[
\Delta^t_p\colon \id_{\cD(\Fp)}\to T_p
\]
which is characterised on homology by 
\[
\begin{tikzcd}[row sep=0, column sep=small]
	(\Delta^t_p)_* &[-20pt] \colon H(V) \arrow[r] & H(T_pV) \\
	& v \arrow[r, |->] & \hat{e}_{(1-p)|v|} \ot (v\ot \overset{p}{\cdots}\ot v).
\end{tikzcd}
\]
\end{definition}
The image of $(\Delta^t_p)_*$ generates $H(T_pV)$ as a $\Fp[u,u^{-1}]$-module with action 
\[
u \Delta^t_*(v) = u (\hat e_{(1-p)|v|} \ot v^{\ot p}) \coloneqq \hat e_{(1-p)|v|+1} \ot v^{\ot p}.
\]
In other words, we have the following:

\begin{lemma}\label{p:tate_iso}
	There is an isomorphism of free \(\Fp[u,u^{-1}]\)-modules:
	\[
	\begin{tikzcd}[row sep=0, column sep=small]
		\Fp[u,u^{-1}] \ot H(V) \arrow[r] & H(T_pV) \\
		\hspace*{13pt} u^k \ot v \arrow[r, |->] & u^k \Delta^t_*(v).
	\end{tikzcd}
	\]
\end{lemma}
\begin{proof}
First, we have that 
\[
H(T_pV)\cong h(\widehat W_p\otimes_{\cyc_p} V^{\otimes p})\cong H(\widehat W_p\otimes_{\cyc_p}H(V)^{\otimes p}).
\]
Now, $H(V)$ is a graded $\Fp$-vector space, so we can consider a basis $\{v_\alpha\}_{\alpha\in A}$ for $H(V)$. Since $H(V)^{\ot p}$ is a permutation module, we can decompose it into orbits. There are two types of orbits, the trivial orbits with a single element and the free orbits with $p$ elements, that is,
\[
H(V)^{\ot p}\cong \left (\,\bigoplus_{\alpha\in A} \Fp\{v_\alpha^{\ot p}\}\right ) \oplus \left (\,\bigoplus_{\text{free orbits}} \Fp[\cyc_p]\right ).
\]
Tensoring over $\cyc_p$ with $\widehat W_p$ gives 
\[
\widehat W_p\ot_{\cyc_p} H(V)^{\ot p}\cong \left (\,\bigoplus_{\alpha\in A}\widehat W_p\ot_{\cyc_p} \Fp\{v_\alpha^{\ot p}\}\right )\oplus \left (\,\bigoplus_{\text {free orbits}} \widehat W_p\right ),
\]
but $\widehat W_p$ is acyclic, and $H(\widehat W_p\ot_{\cyc_p} \Fp)\cong H(T_p\Fp)\cong \Fp[u,u^{-1}]$. It is thus clear that the given map is an isomorphism.
\end{proof}

\begin{lemma}
The algebraic Tate diagonal is $\bF_p$-linear.
\end{lemma}
\begin{proof}
First we check scalar multiplicativity, let $a\in\bF_p$ and $x\in H_n(V)$.
\[
\Delta^t_p(ax)=\hat e_{(1-p)n}\ot(ax)^{\ot p}=a^p(\hat e_{(1-p)n}\ot x^{\ot p})=a\Delta^t_p(x),
\]
where the last equality holds in $\bF_p$ due to Fermat's little theorem.

Now, for additivity, let $x$ and $y\in H_n(V)$. We have
\[
\Delta^t_p(x+y)=\hat e_{(1-p)n}\ot(x+y)^{\ot p}.
\]
The term $(x+y)^{\ot p}$ is a sum of $2^p$ terms, corresponding to the set of all possible words of length $p$ using the letters $x$ and $y$. The cyclic group acts on this set by permutation, and there are two types of orbits: those consisting of a single element and those free with $p$ elements. There are two trivial orbits with a single element, corresponding to $x^{\otimes p}$ and $y^{\otimes p}$ respectively. Let ${\bf w}_1,\dots, {\bf w}_{(2^{p}-2)/p}$ be representatives of the remaining orbits. Thus, we can write
\[
\hat e_{(1-p)n}\ot(x+y)^{\ot p}=\hat e_{(1-p)n}\ot x^{\ot p}+\hat e_{(1-p)n}\ot y^{\ot p}+\sum_{i=1}^{(2^{p}-2)/p}\hat e_{(1-p)n}\ot N{\bf w}_i,
\]
%but 
%\[
%\hat e_{(1-p)n}\ot_{\cyc_p}N{\bf w}_i=N\hat e_{(1-p)n}\ot_{\cyc_p}{\bf w}_i=d\hat e_{(1-p)n+1}\ot_{\cyc_p}{\bf w}_i=d(\hat e_{(1-p)n+1}\ot_{\cyc_p}{\bf w}_i),
%\]
%where the second equality holds because $(1-p)n+1$ is an odd number for $p$ prime (and $d\hat e_j=N\hat e_{j-1}$ for any $j$ if $p=2$), and the last equality holds since $x$ and $y$ are cycles. 
but the elements $\hat e_{(1-p)n}\ot N{\bf w}_i$ lie on the subcomplex of $\widehat W_p\ot  H(V)^{\ot p}$ of free orbits, which is acyclic, meaning that the term $\sum_{i=1}^{(2^{p}-2)/p}\hat e_{(1-p)n}\ot N{\bf w}_i$ vanishes, and so
\[
\Delta^t_p(x+y)=\Delta^t_p(x)+ \Delta^t_p(y).\qedhere
\]
\end{proof}
\subsection{A characterisation of the algebraic Tate diagonal}

Consider the full subcategory  $\Fun^{\mathrm{Ex}}(\cD(\Fp),\cD(\Fp))\subset \Fun(\cD(\Fp),\cD(\Fp))$ of exact functors. The version on $\cD(\Fp)$ of Proposition III.1.2 from [Nikolaus-Scholze] is the following:
\begin{proposition}
For any $F\in \Fun^{\mathrm{Ex}}(\cD(\Fp),\cD(\Fp))$, evaluation at the chain complex $\Fp$ concentrated in degree $0$ induces an equivalence
\[
\mathrm{Map}_{\Fun^{\mathrm{Ex}}(\cD(\Fp),\cD(\Fp))}(\id_{\cD(\Fp)}, F)\longrightarrow \Hom_{\cD(\Fp)}(\Fp,F(\Fp)).
\]
\end{proposition}

Therefore, there is an equivalence 
\[
\mathrm{Map}_{\Fun^{\mathrm{Ex}}(\cD(\Fp),\cD(\Fp))}(\id_{\cD(\Fp)}, T_p)\longrightarrow \Hom_{\cD(\Fp)}(\Fp,T_p(\Fp))\cong\Fp\{\hat e_0\},
\]
and we can characterise the algebraic Tate diagonal in the following way:
\begin{proposition}
The algebraic Tate diagonal $\Delta^t_p$ is the natural transformation 
\[
\Delta^t_p\colon \id_{\cD(\Fp)}\to T_p
\]
that evaluated on the complex $\Fp$ concentrated in degree $0$ corresponds to the inclusion $\Fp\hookrightarrow T_p(\Fp)$ given by $1\mapsto \hat e_0$. 
\martin{This isn't completely canonical tho, it depends on the choice of the isomorphism $T_p(\Fp)_0\cong \Fp\{\hat e_0\}$.} 
\end{proposition}
\subsection{Relations with the spectral Tate diagonal}
Let ${\mathsf {Sp}}$ denote the $\infty$-category of spectra. In [Nikolaus-Scholze], they define the following:
\begin{definition}
Let $X$ be a spectrum. 
Let  $T^{\mathsf {Sp}}_p$ denote the functor $\mathsf{Sp}\to \mathsf{Sp}$ given by
\[
T^{\mathsf {Sp}}_p(X)\coloneqq (X^{\wedge p})^{t\cyc_p}=\cofib (Nm\colon (X\wedge \overset{p}{\dots} \wedge X)_{h\cyc_p}\to (X\wedge \overset{p}{\dots} \wedge X)^{h\cyc_p}),
\]
the Tate construction of $X^{\wedge p}$, with $\cyc_p$-action given by permuting the factors.
\end{definition}
They prove that it is an exact functor and give
\begin{proposition}
For any $F\in \Fun^{\mathrm{Ex}}(\mathsf{Sp},\mathsf{Sp})$, evaluation at the sphere spectrum $\bS$ induces an equivalence
\[
\mathrm{Map}_{\Fun^{\mathrm{Ex}}({{\mathsf{Sp}}},{{\mathsf{Sp}}})}(\id_{{{\mathsf{Sp}}}}, F)\longrightarrow \Hom_{\mathsf{Sp}}(\bS,F(\bS))=\Omega^\infty F(\bS).
\]
\end{proposition}
Thus, the space of natural transformations $\id_{\mathsf{Sp}}\to T_p^{\mathsf{Sp}}$ is equivalent to $\Hom_{\mathsf{Sp}}(\bS,T_p^{\mathsf{Sp}}(\bS))=\Omega^\infty T^{\mathsf {Sp}}_p(\bS)$. This allows them to define
\begin{definition}
The \defn{Tate diagonal} is the natural transformation $\Delta_p\colon \id_{\mathsf{Sp}}\to T_p^{\mathsf{Sp}}$ that corresponds to the composition 
\[
\bS\to\bS^{h\cyc_p}\to \bS^{t\cyc_p}=T_p^{\mathsf{Sp}}(\bS).
\]
\end{definition}

\subsection{Monoidality}

The identity functor $\id_{\cD (\Fp)}$ is a lax monoidal functor: for chain complexes $C$ and $D$, there is a natural morphism $\kappa$ in $\cD(\Fp)$ that is characterised in homology by the Künneth isomorphism \martin{is this correctly formulated? Or rather $\kappa$ is the identity, which is characterised in homology by the kunneth isomorphism? Thus the identity is monoidal, not lax monoidal.}
\[
\kappa_* \colon H(C)\otimes H(D)\longrightarrow H(C\otimes D).
\]

The functor $T_p$ is also a lax monoidal functor. Indeed, there is a natural morphism characterised by
\[
\mu_*\circ \kappa_*\colon H\big(\widehat W_p\otimes_{\cyc_p} C^{\otimes p}\big)\otimes H\big(\widehat W_p\otimes_{\cyc_p} D^{\otimes p}\big)
\longrightarrow H\big(\widehat W_p\otimes_{\cyc_p} (C\otimes D)^{\otimes p}\big),
\]
where
\[
\mu_*\colon H\big((\widehat W_p\otimes_{\cyc_p} C^{\otimes p})\otimes (\widehat W_p\otimes_{\cyc_p} D^{\otimes p})\big)
\longrightarrow H\big(\widehat W_p\otimes_{\cyc_p} (C\otimes D)^{\otimes p}\big)
\]
is induced by the assignment
\[
[(\rho^a e_i\otimes_{\cyc_p} \mathbf{c})\otimes (\rho^b e_j\otimes_{\cyc_p} \mathbf{d})]
\longmapsto
[\,e_{i+j}\otimes_{\cyc_p} \mathrm{Sh}(\rho^{-a}\mathbf{c}, \rho^{-b}\mathbf{d})\,],
\]
for $\mathbf{c}=c_1\otimes\cdots\otimes c_p \in C^{\otimes p}$ and $\mathbf{d}=d_1\otimes\cdots\otimes d_p \in D^{\otimes p}$. Here $\mathrm{Sh}$ denotes the shuffle map $C^{\otimes p}\times D^{\otimes p}\to(C\otimes D)^{\otimes p}$ given by
\[
((c_1\otimes\cdots\otimes c_p), (d_1\otimes\cdots\otimes d_p))
\longmapsto
(c_1\otimes d_1)\otimes\cdots\otimes (c_p\otimes d_p).
\]

The following diagram commutes for all chain complexes $C$ and $D$:
\begin{center}
\begin{tikzpicture}
\node (a) at (-3,2){$H(C)\otimes H(D)$};
\node (b) at (4,2){$H(\widehat W_p\otimes_{C_p} C^{\otimes p})\otimes H(\widehat W_p\otimes_{C_p} D^{\otimes p})$};
\node (c) at (-3,0){$H(C\otimes D)$};
\node (d) at (4,0){$H(\widehat W_p\otimes_{C_p} (C\otimes D)^{\otimes p})$};
%
\draw[->](a) -- node[above]{$(\Delta^t_p)_*\otimes (\Delta^t_p)_*$}(b);
\draw[->](a)--node[right]{$\kappa_*$}(c);
\draw[->](b)--node[right]{$\mu_*\circ\kappa_*$}(d);
\draw[->](c)--node[above]{$(\Delta^t_p)_*\circ \otimes$}(d);

\end{tikzpicture}
\end{center}


\begin{lemma}
The algebraic Tate diagonal defines a monoidal natural transformation
\[
\Delta^t_p\colon \id_{\cD (\Fp)} \longrightarrow T_p.
\]
\end{lemma}
